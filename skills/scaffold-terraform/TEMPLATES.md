### Makefile (stacks repo)

Every tool invocation goes through `$(MISE)`, so the versions pinned in `.mise.toml` are the versions that run, locally and in CI. Targets that need a tool depend on `install` (which depends on `preflight`). Logic longer than one command lives in `scripts/`.

```makefile
.DEFAULT_GOAL := help
SHELL := $(shell which bash)
MISE := mise exec --

help: ## Show this help
	@fgrep -h "##" $(MAKEFILE_LIST) | fgrep -v fgrep | sed -e 's/\\$$//' | sed -e 's/##//'

preflight: ## verify the non-mise tools the checks rely on (CI runners, slim images)
	bash scripts/preflight.sh

install: preflight ## install the tool versions pinned in .mise.toml
	mise trust --yes
	mise install

hooks: install ## install the git hooks (pre-commit runs make static)
	$(MISE) pre-commit install

fmt: install ## format terraform and hcl files
	$(MISE) terraform fmt -recursive
	$(MISE) terragrunt hcl fmt

fmt-check: install ## check formatting without modifying
	$(MISE) terraform fmt -recursive -check
	$(MISE) terragrunt hcl fmt --check

validate: install ## init (no backend) and validate every module
	$(MISE) scripts/validate.sh

lint: install ## run tflint
	$(MISE) tflint --init --config "$(CURDIR)/.tflint.hcl"
	$(MISE) tflint --recursive --config "$(CURDIR)/.tflint.hcl"

security: install ## scan the working tree for secrets (betterleaks)
	$(MISE) betterleaks dir .

pins: ## fail on version ranges (~>, >=) in .tf/.hcl files
	bash scripts/check-pins.sh

shellcheck: install ## check committed shell scripts
	git ls-files -z '*.sh' | xargs -0 -r $(MISE) shellcheck --severity=error --format=gcc

unit: install ## terraform test in every module with tests/unit.tftest.hcl (no credentials)
	$(MISE) scripts/unit.sh unit

unit-update-golden: ## no golden files in terraform modules
	@echo "no golden files"

integration: install ## terraform test integration suites (requires credentials)
	$(MISE) scripts/unit.sh integration

static-check: preflight fmt-check validate lint security pins shellcheck unit ## run all static checks (CI)

static: preflight fmt validate lint security pins shellcheck unit ## run all static checks with auto-format

.PHONY: help preflight install hooks fmt fmt-check validate lint security pins shellcheck unit unit-update-golden integration static-check static
```

`static` and `static-check` list every available check plus `shellcheck`; when a new check is added to the project, add it to both. Modules declare no providers and no `provider_aws_*` variables: the envs repo's generated `versions.tf` supplies both, and a module-level `versions.tf` would collide with it. `validate` and `unit` therefore resolve the provider unpinned; pinning lives in the envs repo.

`scripts/validate.sh`:
```bash
#!/usr/bin/env bash
# Run terraform init (no backend) and validate in every directory with committed .tf files.
# Run via: mise exec -- scripts/validate.sh
set -euo pipefail

root=$(git rev-parse --show-toplevel)
status=0
while read -r dir; do
  echo "== validate ${dir}"
  terraform -chdir="${root}/${dir}" init -backend=false -input=false >/dev/null || { status=1; continue; }
  terraform -chdir="${root}/${dir}" validate || status=1
done < <(git -C "${root}" ls-files '*.tf' | xargs -r -n1 dirname | sort -u)
exit "${status}"
```

`scripts/unit.sh`:
```bash
#!/usr/bin/env bash
# Run terraform test in every module that has tests/<suite>.tftest.hcl (suite: unit or integration).
# Run via: mise exec -- scripts/unit.sh [unit|integration]
set -euo pipefail

suite=${1:-unit}
root=$(git rev-parse --show-toplevel)
status=0
while read -r dir; do
  echo "== ${suite} ${dir}"
  terraform -chdir="${root}/${dir}" init -backend=false -input=false >/dev/null || { status=1; continue; }
  terraform -chdir="${root}/${dir}" test -filter="tests/${suite}.tftest.hcl" || status=1
done < <(git -C "${root}" ls-files "*/tests/${suite}.tftest.hcl" | xargs -r -n1 dirname | xargs -r -n1 dirname | sort -u)
exit "${status}"
```

Make every script executable (`chmod +x scripts/*.sh`) and `git add` it, so `shellcheck` sees it.

`stacks/environment/example/terragrunt.hcl` (a unit; the envs repo references the folder):
```hcl
terraform {
  source = "${local.common_env.iac_modules_root}/example?ref=${local.common_env.iac_release_tag}"
}

locals {
  env        = yamldecode(file(mark_as_read(abspath(find_in_parent_folders(values.env_filename)))))
  common_env = yamldecode(file(mark_as_read(abspath(find_in_parent_folders(values.env_common_filename)))))
  account    = yamldecode(file(mark_as_read(abspath(find_in_parent_folders(values.account_filename)))))
}

inputs = {
  buckets = local.env.example.buckets
  tags    = local.common_env.base-tags

  provider_aws_region     = local.common_env.primary_region
  provider_aws_account_id = local.account.aws_account_id
}

include "root" {
  path = find_in_parent_folders("root.hcl")
}
```

The unit contract: every unit reads its config from the three yaml files named in `values` (`env_filename`, `env_common_filename`, `account_filename`), sources its module at the same release tag as the stack (`iac_release_tag`), and passes `provider_aws_region` and `provider_aws_account_id`. A multi-unit stack is a `stacks/environment/<name>/terragrunt.stack.hcl` whose `unit` blocks use `source = "./<unit>"` and pass the three values through (`env_filename = values.env_filename`, and so on).

`README.md` for the stacks repo:
```markdown
# {{REPO_NAME}}

Terragrunt stacks and Terraform modules consumed by environment repos (`{{ENVS_REPO_NAME}}`) by release tag.

## Changing a stack or module
1. Branch here and make the change.
2. In the envs repo, point `iac_release_tag` in `env-common.yaml` at your branch to test with `terragrunt stack generate` and `terragrunt plan`.
3. Open the PR here. `make static` must pass.
4. Merge, update `CHANGELOG.md`, and tag the release (`vX.Y.Z`).
5. In the envs repo, set `iac_release_tag` to the new tag and roll out.

## Choosing the release tag
- **Patch**: changes inside existing files; backwards compatible; no state moves, no destroys; small new optional yaml keys.
- **Minor**: new modules or functionality; removals; breaking yaml changes; destroys or removed folders; core service or module version updates. No manual state moves.
- **Major**: cross-cutting changes touching more than one module; major env yaml refactors; manual state migrations or manual AWS changes; provider or Kubernetes version updates.
```

### scripts/preflight.sh and scripts/check-pins.sh

Both repo types. `preflight.sh` reports missing non-mise tools up front (a bare CI image gets one clear message instead of a mid-run failure). `make` and `bash` cannot be checked here: `make` must exist to run the Makefile, and `SHELL := $(shell which bash)` is evaluated at parse time.

```bash
#!/usr/bin/env bash
# Verify the tools make static relies on that mise does not manage.
set -euo pipefail

missing=()
for tool in mise git bash xargs grep; do
  command -v "${tool}" >/dev/null 2>&1 || missing+=("${tool}")
done
if [ "${#missing[@]}" -gt 0 ]; then
  echo "ERROR: missing required tools: ${missing[*]}" >&2
  exit 1
fi

git rev-parse --is-inside-work-tree >/dev/null 2>&1 || { echo "ERROR: not inside a git work tree" >&2; exit 1; }
# BusyBox variants can lack these flags
printf '' | xargs -0 -r true 2>/dev/null || { echo "ERROR: xargs lacks -0/-r" >&2; exit 1; }
echo ab | grep -qE 'a(b)' 2>/dev/null || { echo "ERROR: grep lacks -E" >&2; exit 1; }
```

```bash
#!/usr/bin/env bash
# Fail on version ranges (~>, >=) in .tf and .hcl files, tracked or not yet committed.
set -euo pipefail

cd "$(git rev-parse --show-toplevel)"
files=()
while IFS= read -r -d '' f; do
  files+=("${f}")
done < <(git ls-files -z -co --exclude-standard -- '*.tf' '*.hcl')

if [ "${#files[@]}" -eq 0 ]; then
  echo "no .tf/.hcl files to check"
  exit 0
fi
if grep -nE '(~>|>=)' "${files[@]}"; then
  echo "ERROR: version ranges found; pin exact versions" >&2
  exit 1
fi
echo "All version pins exact"
```

The array guard matters: `xargs -r` exits 0 on empty input, which would let the check pass on nothing. A `>=` inside a Terragrunt expression is a false positive; rewrite the expression.

### State bootstrap: scripts/bootstrap-state.sh

Envs repo only. `make bootstrap-state` creates the remote-state S3 bucket once per AWS account, before the first `init`. It is idempotent: re-running reconciles settings and skips what exists. It needs credentials, so it stays out of `static`. Naming convention: bucket `<prefix>-terraform-state-<account-id>` (account-suffixed, globally unique). Locking is S3-native (`use_lockfile = true`, a `.tflock` object beside the state), so there is no DynamoDB table; the DynamoDB lock backend is deprecated.

Best-practice settings applied: bucket versioning, default SSE (AES256 with bucket key), all public access blocked, bucket-owner-enforced ownership (ACLs off), TLS-only bucket policy, noncurrent versions expire after 90 days, incomplete multipart uploads abort after 7 days. Versioning also protects the state and lock objects. The script refuses to run when the caller's account differs from the expected one.

```bash
#!/usr/bin/env bash
# Create the Terraform remote-state S3 bucket if missing (idempotent).
# Usage: scripts/bootstrap-state.sh ACCOUNT_ID BUCKET REGION
set -euo pipefail

if [ $# -ne 3 ]; then
  echo "Usage: $0 ACCOUNT_ID BUCKET REGION" >&2
  exit 2
fi
account_id=$1
bucket=$2
region=$3

caller=$(aws sts get-caller-identity --query Account --output text)
if [ "${caller}" != "${account_id}" ]; then
  echo "ERROR: credentials are for account ${caller}, expected ${account_id}" >&2
  exit 1
fi

echo "== bucket ${bucket} (${region})"
if aws s3api head-bucket --bucket "${bucket}" 2>/dev/null; then
  echo "exists"
else
  if [ "${region}" = "us-east-1" ]; then
    aws s3api create-bucket --bucket "${bucket}" --region "${region}"
  else
    aws s3api create-bucket --bucket "${bucket}" --region "${region}" \
      --create-bucket-configuration "LocationConstraint=${region}"
  fi
fi

aws s3api put-bucket-versioning --bucket "${bucket}" \
  --versioning-configuration Status=Enabled
aws s3api put-bucket-encryption --bucket "${bucket}" \
  --server-side-encryption-configuration \
  '{"Rules":[{"ApplyServerSideEncryptionByDefault":{"SSEAlgorithm":"AES256"},"BucketKeyEnabled":true}]}'
aws s3api put-public-access-block --bucket "${bucket}" \
  --public-access-block-configuration \
  BlockPublicAcls=true,IgnorePublicAcls=true,BlockPublicPolicy=true,RestrictPublicBuckets=true
aws s3api put-bucket-ownership-controls --bucket "${bucket}" \
  --ownership-controls 'Rules=[{ObjectOwnership=BucketOwnerEnforced}]'
aws s3api put-bucket-policy --bucket "${bucket}" --policy "$(cat <<POLICY
{
  "Version": "2012-10-17",
  "Statement": [{
    "Sid": "DenyInsecureTransport",
    "Effect": "Deny",
    "Principal": "*",
    "Action": "s3:*",
    "Resource": ["arn:aws:s3:::${bucket}", "arn:aws:s3:::${bucket}/*"],
    "Condition": {"Bool": {"aws:SecureTransport": "false"}}
  }]
}
POLICY
)"
aws s3api put-bucket-lifecycle-configuration --bucket "${bucket}" \
  --lifecycle-configuration '{"Rules":[{"ID":"expire-noncurrent-and-abort-multipart","Status":"Enabled","Filter":{},"NoncurrentVersionExpiration":{"NoncurrentDays":90},"AbortIncompleteMultipartUpload":{"DaysAfterInitiation":7}}]}'

echo "state backend ready: s3://${bucket}"
```

## Envs repo

An envs repo holds values only: no module code. One **account folder** per AWS account (`<account>/`) holds `account.yaml`, `root.hcl`, `providers/provider_base.hcl`, `.terraform-version`, `.terragrunt-version`. The **environment folder** holds `env-common.yaml`, the `env-*.yaml` domain files, `iam-policies/*.json` and `terragrunt.stack.hcl`. When an account hosts one environment the two are the same folder (`claude-code/`); when it hosts several, each is a subfolder (`bedrock-dev/dev/`). `find_in_parent_folders` resolves the account-level files from either shape.

### scripts/check-layout.sh

Enforces the envs-repo pattern: no `.tf` files, each account folder pins both tool versions and has a `root.hcl`, `iac_release_tag` is an exact `vX.Y.Z` (a feature-branch ref is for local testing and fails the check, so it cannot merge), every `source` is `?ref=` pinned, and every `unit`/`stack` block sets `no_dot_terragrunt_stack = true` plus `env_filename`, `env_common_filename`, `account_filename`. The `aws-*-pipelines-bootstrap` folders are exempt from the contract checks.

```bash
#!/usr/bin/env bash
# Enforce the envs-repo pattern: values only, units pinned by release tag, one values contract.
set -euo pipefail

cd "$(git rev-parse --show-toplevel)"
status=0
fail() {
  echo "ERROR: $*" >&2
  status=1
}

# Module code lives in the stacks repo
while IFS= read -r f; do
  fail "${f}: .tf files belong in the stacks repo"
done < <(git ls-files -co --exclude-standard -- '*.tf')

# Every account folder pins its tool versions
while IFS= read -r account; do
  dir=$(dirname "${account}")
  case "${dir}" in aws-*-pipelines-bootstrap) continue ;; esac
  for pin in .terraform-version .terragrunt-version; do
    [ -f "${dir}/${pin}" ] || fail "${dir}: missing ${pin}"
  done
  [ -f "${dir}/root.hcl" ] || fail "${dir}: missing root.hcl"
done < <(git ls-files -co --exclude-standard -- '*/account.yaml')

# Every environment pins the stacks repo by release tag
while IFS= read -r common; do
  grep -qE '^iac_release_tag: "v[0-9]+\.[0-9]+\.[0-9]+"' "${common}" ||
    fail "${common}: iac_release_tag must be an exact vX.Y.Z release tag (feature refs are for local testing only)"
done < <(git ls-files -co --exclude-standard -- '*env-common.yaml')

# Every stack file: pinned sources and the three-value contract on each block
while IFS= read -r stack; do
  if grep -E '^\s*source\s*=' "${stack}" | grep -qv '?ref='; then
    fail "${stack}: every source needs ?ref=<release tag>"
  fi
  case "${stack}" in aws-*-pipelines-bootstrap/*) continue ;; esac
  blocks=$(grep -cE '^(unit|stack) "' "${stack}" || true)
  for key in no_dot_terragrunt_stack env_filename env_common_filename account_filename; do
    seen=$(grep -cE "^\s*${key}\s*=\s*(true|local\.|values\.)" "${stack}" || true)
    [ "${seen}" -eq "${blocks}" ] ||
      fail "${stack}: ${blocks} unit/stack blocks but ${seen} ${key} entries"
  done
done < <(git ls-files -co --exclude-standard -- '*terragrunt.stack.hcl')

[ "${status}" -eq 0 ] && echo "Layout OK"
exit "${status}"
```

`scripts/for-each-env.sh`:
```bash
#!/usr/bin/env bash
# Run a command inside every environment folder (a top-level folder holding account.yaml).
# Usage: scripts/for-each-env.sh COMMAND [ARGS...]
set -euo pipefail

root=$(git rev-parse --show-toplevel)
status=0
for account in "${root}"/*/account.yaml; do
  env_dir=$(dirname "${account}")
  case "$(basename "${env_dir}")" in aws-*-pipelines-bootstrap) continue ;; esac
  echo "== ${env_dir#"${root}"/}"
  (cd "${env_dir}" && "$@") || status=1
done
exit "${status}"
```

## .pre-commit-config.yaml

The git hook runs `make static`, which includes the betterleaks secret scan, shellcheck and the terraform checks. `make hooks` installs it.

```yaml
repos:
  - repo: local

## Keep `make static` clean

Enforced by the `scaffold-terraform` skill:

1. **Pinned versions**: terraform, terragrunt, providers, modules, and GitHub Actions use exact versions (tools in `.mise.toml`).

## Terragrunt pattern

{{REPO_ROLE_PARAGRAPH}}

- Config is yaml, not HCL: `account.yaml` (account id, state bucket), `env-common.yaml` (`iac_release_tag`, regions, `base-tags`), `env-<domain>.yaml` per domain, IAM policies as JSON under `iam-policies/`.
- Units and stacks are pinned by `?ref=<iac_release_tag>`; a feature-branch ref is for local testing and is replaced with a release tag before merge.
- Every `unit`/`stack` block sets `no_dot_terragrunt_stack = true` and passes `env_filename`, `env_common_filename`, `account_filename`.
- Stack output directories are generated and gitignored. Run `terragrunt stack generate`, then `terragrunt plan` inside the generated folder.
- Choose the release tag by blast radius: patch (backwards compatible), minor (new or removed functionality, breaking yaml), major (cross-cutting, state migrations, provider upgrades).
```

`{{REPO_ROLE_PARAGRAPH}}` is one sentence naming the role: envs repo ("This repo holds environment values only; modules and stacks live in `{{STACKS_REPO}}`.") or stacks repo ("This repo holds modules and stacks consumed by `{{ENVS_REPO}}` by release tag; it holds no environment values.").

### scripts/detect-env.sh

Plan and apply both call this, so the logic is shellchecked and lives in one place. Arguments are passed to `git diff --name-only` (plan: `origin/$BASE_REF...HEAD`; apply: `$BEFORE $AFTER`). `PROD_REGEX` defaults to the prod folder prefix from the interview.

```bash
#!/usr/bin/env bash
# Detect the environment(s) a change touches and refuse to mix prod with non-prod.
# Usage: scripts/detect-env.sh GIT_DIFF_ARGS...   (writes environment=... to $GITHUB_OUTPUT when set)
set -euo pipefail

prod_regex="${PROD_REGEX:-{{PROD_REGEX}}}"

changed_files=$(git diff --name-only "$@")
top_level_folders=$(echo "${changed_files}" | grep -v '^$' | grep -vE '^(README\.md|\.gitignore)$|^(\.github|\.gruntwork|aws-|scripts)' | cut -d'/' -f1 | sort -u || true)

environment=""
for folder in ${top_level_folders}; do
  environment="${environment:+${environment}-}${folder}"
done
environment="${environment:-unknown}"
echo "Detected environment: ${environment}"
echo "environment=${environment}" >> "${GITHUB_OUTPUT:-/dev/stdout}"

prod_folders=$(echo "${top_level_folders}" | grep -E "${prod_regex}" || true)
nonprod_folders=$(echo "${top_level_folders}" | grep -vE "${prod_regex}" | grep -v '^$' || true)
if [ -n "${prod_folders}" ] && [ -n "${nonprod_folders}" ]; then
  echo "ERROR: prod and non-prod folders changed together; submit production changes in a separate PR." >&2
  exit 1
fi
```

name: Pipelines Plan

    runs-on: ubuntu-latest

      environment: ${{ steps.detect-env.outputs.environment }}
    steps:
      - name: Checkout code

          fetch-depth: 0

    runs-on: ubuntu-latest

      group: terragrunt-plan-env-${{ needs.DetectEnvironment.outputs.environment }}
